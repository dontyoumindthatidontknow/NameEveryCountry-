    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = button

-- DRAGGING: MOUSE + MOBILE TOUCH
local dragging = false
local dragStart
local startPosition
local dragInput

title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = border.Position
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end

    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input == dragInput then

        local delta = input.Position - dragStart

        border.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input == dragInput then
        dragging = false
        dragInput = nil
    end
end)

-- TEXTBOX DETECTION
local function findFocusedBox()
    for _, object in ipairs(playerGui:GetDescendants()) do
        if object:IsA("TextBox")
            and object.Visible
            and object:IsFocused() then

            return object
        end
    end

    return nil
end

-- TYPE COUNTRIES
local function runSequence()
    if typing then return end

    typing = true
    local mySession = session

    task.spawn(function()
        while enabled and mySession == session do

            local box = activeBox

            if not box or not box.Parent then
                box = findFocusedBox()
                activeBox = box
            end

            if box and box:IsFocused() then

                local country = countries[index]

                if not country then
                    enabled = false
                    status.Text = "Status: FINISHED!"
                    break
                end

                -- Set text in an authorized TextBox
                box.Text = country

                -- Submit through Roblox's supported TextBox API.
                -- FocusLost receives enterPressed = true.
                box:ReleaseFocus(true)

                index += 1

                progress.Text =
                    "Countries: " .. (index - 1) .. " / " .. #countries

                if index > #countries then
                    enabled = false
                    status.Text = "Status: FINISHED!"
                    status.TextColor3 = Color3.fromRGB(100, 255, 130)
                    break
                end

                task.wait(COOLDOWN)

                -- Resume only if the same textbox still exists.
                if enabled and mySession == session
                    and box.Parent then
                    activeBox = box
                    box:CaptureFocus()
                end

            else
                activeBox = findFocusedBox()
                task.wait(0.1)
            end
        end

        typing = false
    end)
end

-- START
startButton.MouseButton1Click:Connect(function()
    if enabled then return end

    enabled = true
    session += 1

    status.Text = "Status: WAITING FOR TEXTBOX"
    status.TextColor3 = Color3.fromRGB(255, 210, 80)

    activeBox = findFocusedBox()
    runSequence()
end)

-- REPEAT
repeatButton.MouseButton1Click:Connect(function()
    enabled = false
    session += 1
    index = 1
    activeBox = nil

    progress.Text = "Countries: 0 / " .. #countries

    task.wait(0.1)

    enabled = true
    status.Text = "Status: RESTARTING"
    status.TextColor3 = Color3.fromRGB(100, 180, 255)

    activeBox = findFocusedBox()
    runSequence()
end)

-- STOP
stopButton.MouseButton1Click:Connect(function()
    enabled = false
    session += 1
    activeBox = nil

    status.Text = "Status: STOPPED"
    status.TextColor3 = Color3.fromRGB(255, 100, 100)
end)
